#include <stdio.h>

int main() {
    int n, bt[10], p[10], wt[10], tat[10];
    int i, j, temp;
    float avgwt = 0, avgtat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        p[i] = i + 1;
        printf("Enter BT and Priority of P%d: ", i+1);
        scanf("%d %d", &bt[i], &p[i+5]);
    }

    for(i = 0; i < n; i++)
        for(j = i+1; j < n; j++)
            if(p[i+5] > p[j+5]) {
                temp = bt[i]; bt[i] = bt[j]; bt[j] = temp;
                temp = p[i]; p[i] = p[j]; p[j] = temp;
                temp = p[i+5]; p[i+5] = p[j+5]; p[j+5] = temp;
            }

    wt[0] = 0;

    for(i = 1; i < n; i++)
        wt[i] = wt[i-1] + bt[i-1];

    for(i = 0; i < n; i++) {
        tat[i] = wt[i] + bt[i];
        avgwt += wt[i];
        avgtat += tat[i];
    }

    printf("\nP\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\n", p[i], bt[i], wt[i], tat[i]);

    printf("Average WT = %.2f\n", avgwt/n);
    printf("Average TAT = %.2f\n", avgtat/n);

    return 0;
}
