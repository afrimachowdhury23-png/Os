#include <stdio.h>

int main() {
    int n, at[10], bt[10], rt[10], wt[10], tat[10];
    int i, time = 0, completed = 0;
    float avgwt = 0, avgtat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("Enter AT and BT of P%d: ", i+1);
        scanf("%d %d", &at[i], &bt[i]);
        rt[i] = bt[i];
    }

    while(completed < n) {
        int shortest = -1;

        for(i = 0; i < n; i++)
            if(at[i] <= time && rt[i] > 0 &&
              (shortest == -1 || rt[i] < rt[shortest]))
                shortest = i;

        if(shortest == -1) {
            time++;
            continue;
        }

        rt[shortest]--;
        time++;

        if(rt[shortest] == 0) {
            completed++;
            tat[shortest] = time - at[shortest];
            wt[shortest] = tat[shortest] - bt[shortest];
            avgwt += wt[shortest];
            avgtat += tat[shortest];
        }
    }

    printf("\nP\tWT\tTAT\n");
    for(i = 0; i < n; i++)
        printf("P%d\t%d\t%d\n", i+1, wt[i], tat[i]);

    printf("Average WT = %.2f\n", avgwt/n);
    printf("Average TAT = %.2f\n", avgtat/n);

    return 0;
}
